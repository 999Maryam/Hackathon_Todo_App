To access your application:

{{- if eq .Values.service.type "ClusterIP" }}

1. Forward the port to your local machine:

   kubectl port-forward svc/{{ .Release.Name }}-todo-backend {{ .Values.service.port }}:{{ .Values.service.port }}

2. Access the application at:

   http://localhost:{{ .Values.service.port }}

{{- else if eq .Values.service.type "NodePort" }}

1. Get the application URL:

   export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ .Release.Name }}-todo-backend)
   export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
   echo http://$NODE_IP:$NODE_PORT

{{- else if eq .Values.service.type "LoadBalancer" }}

1. Get the application URL:

   NOTE: It may take a few minutes for the LoadBalancer IP to be available.

   export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ .Release.Name }}-todo-backend --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
   echo http://$SERVICE_IP:{{ .Values.service.port }}

{{- end }}
